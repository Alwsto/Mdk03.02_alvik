using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using GuestPass.Api;

[ApiController]
[Route("api/passes")]
public sealed class VulnerableController(PassDbContext db) : ControllerBase
{
    [HttpGet("{id}")]
    public async Task<IActionResult> Get(int id)
    {
        var pass = await db.Passes.FirstOrDefaultAsync(x => x.Id == id);
        return pass is null ? NotFound() : Ok(pass);
    }
}
